הָאֲבָנִים מְֽדַבְּרוֹת — מָמִילָא

בְּרוּכִים הַבָּאִים לַפּוֹ֫דְקַאסְט הָאֲבָנִים מְֽדַבְּרוֹת. הַיּוֹם אֲנַ֫חְנוּ מְֽטַיְּלִים בִּירוּשָׁלַ֫יִם, בִּשְׂדֵרָה שֶׁכֻּלָּם מַכִּירִים: מָמִילָא. הַחֲנֻיּוֹת, בָּתֵּי הַקָּפֶה, הַנּוֹף אֶל הַחוֹמוֹת… וּבְכׇל זֹאת, כִּמְעַט אַף אֶחָד לֹא יוֹדֵ֫עַ מָה הָיָה כָּאן קֹ֫דֶם. אַתֶּם רוֹאִים קַנְיוֹן. אֲנִי רוֹאֶה עִיר שֶׁנֶּעֶלְמָה.

נַתְחִיל מֵהַמַּ֫יִם. מִמַּעֲרָב לַשְּׂדֵרָה, בְּֽגַן הָעַצְמָאוּת, נִמְצֵאת בְּרֵכָה גְּדוֹלָה: בְּרֵכַת מָמִילָא. זֶ֫הוּ מַאֲגַר מַ֫יִם עַתִּיק, כְּנִרְאֶה מִימֵי הוֹרְדוֹס. הוּא הֶחֱזִיק כִּשְׁלוֹשִׁים אֶ֫לֶף מֶ֫טֶר מְֽעֻקָּב שֶׁל מַ֫יִם, וּתְֽעָלָה תַּת־קַרְקָעִית חִבְּרָה אוֹתוֹ לִבְרֵכַת חִזְקִיָּהוּ, בְּֽתוֹךְ הָעִיר הָעַתִּיקָה.

סְבִיב הַבְּרֵכָה הַזֹּאת מִתְרַחֵשׁ אֶחָד הַסִּפּוּרִים הַקָּשִׁים שֶׁל הַמָּקוֹם. בִּשְׁנַת שֵׁשׁ מֵאוֹת וְאַרְבַּע־עֶשְׂרֵה כּוֹבְשִׁים הַפַּרְסִים אֶת יְֽרוּשָׁלַ֫יִם, וְנוֹצְרִים נִטְבָּחִים לְֽיַד הַבְּרֵכָה. כְּרוֹנִיקָן בִּיזַ֫נְטִי בְּֽשֵׁם אַסְטְרָטֵגְיוֹס מְֽתָאֵר אֶת הַטֶּ֫בַח, וּמַאֲשִׁים אֶת הַיְּֽהוּדִים בִּנְקָמָה. חָשׁוּב לוֹמַר: זוֹ הַגִּרְסָה שֶׁל מָקוֹר אֶחָד בִּלְבַד, נָזִיר נוֹצְרִי מֵאוֹתָהּ תְּקוּפָה. אֲבָל הָאַרְכֵאוֹלוֹגְיָה נָתְנָה לוֹ הֵד מַפְתִּ֫יעַ.

בְּֽסוֹף שְׁנוֹת הַשְּׁמוֹנִים שֶׁל הַמֵּאָה הָעֶשְׂרִים הִתְחִילוּ לַחְפֹּר כָּאן אֶת הַחַנְיוֹן שֶׁל הַקַּנְיוֹן הֶעָתִידִי. הַפּוֹעֲלִים נִתְקְֽלוּ בִּמְעָרָה טִבְעִית, בְּֽאֹ֫רֶךְ שֶׁל כִּשְׁנֵים־עָשָׂר מֶ֫טֶר, מְֽלֵאָה בַּעֲצָמוֹת שֶׁל מֵאוֹת אֲנָשִׁים. רַבִּים מֵהֶם הָיוּ צְֽעִירִים, גְּבָרִים וְנָשִׁים.

לִפְנֵי הַמְּֽעָרָה עָמְדָה קַפֶּ֫לָּה קְטַנָּה, עִם רִצְפַּת פְּסֵיפָס. בָּאַ֫פְּסִיס, הַגֻּמְחָה שֶׁבַּקִּיר, נִשְׁאֲרוּ שְׂרִידֵי צִיּוּר: מַלְאָךְ הַפּוֹרֵשׂ אֶת יָדָיו. הָאַרְכֵאוֹלוֹג שֶׁנִּהֵל אֶת הַחֲפִירָה, פְּרוֹפֶ֫סוֹר רוֹנִי רַיְךְ, הִמְשִׁיךְ בַּעֲבוֹדָה לַמְרוֹת הַטְרָדוֹת שֶׁל מַפְגִּינִים חֲרֵדִים. לְֽדַעְתּוֹ, אֵ֫לֶּה כְּנִרְאֶה קׇרְבְּֽנוֹת שְׁנַת שֵׁשׁ מֵאוֹת וְאַרְבַּע־עֶשְׂרֵה.

וְֽהַיּוֹם? הַקַּפֶּ֫לָּה נִמְצֵאת מִתַּ֫חַת לַחַנְיוֹן. כְּֽשֶׁאַתֶּם מַחְנִים אֶת הָרֶ֫כֶב בְּֽמָמִילָא, אוּלַי אַתֶּם עוֹמְדִים בְּֽדִיּוּק מֵעָלֶ֫יהָ.

נִקְפֹּץ לְֽסוֹף הַמֵּאָה הַתְּשַׁע־עֶשְׂרֵה. הַמָּקוֹם כִּמְעַט רֵיק, בְּֽעִקָּר צֹ֫מֶת דְּרָכִים מוּל שַׁ֫עַר יָפוֹ. אֶחָד הַבִּנְיָנִים הָרִאשׁוֹנִים כָּאן הוּא מִנְזָר שֶׁל הָאֲחָיוֹת מִמִּסְדַּר בְּנוֹת הַצְּדָקָה שֶׁל סָן וַנְסָן דֶּה פּוֹל. הֵן הִגִּיעוּ בִּשְׁנוֹת הַשְּׁמוֹנִים שֶׁל הַמֵּאָה הַתְּשַׁע־עֶשְׂרֵה, בַּתְּֽחִלָּה כְּֽדֵי לְֽטַפֵּל בְּֽחוֹלֵי צָרַ֫עַת.

וְֽהִנֵּה הַהַפְתָּעָה: דַּוְקָא הַנְּֽזִירוֹת הֵן שֶׁהִתְחִילוּ כָּאן אֶת הַמִּסְחָר. הֵן בָּנוּ שׁוּרַת חֲנֻיּוֹת מוּל הַמִּנְזָר, וּמִדְּמֵי הַשְּׂכִירוּת הֶחֱזִיקוּ אֶת הַמִּנְזָר. כָּךְ שֶׁחֲנֻיּוֹת הַיֻּקְרָה שֶׁל הַיּוֹם הֵן, בְּֽמִדָּה מְֽסֻיֶּמֶת, הַיּוֹרְשׁוֹת שֶׁל הָרַעְיוֹן שֶׁלָּהֶן. הַמִּנְזָר עוֹמֵד עַד הַיּוֹם בִּמְקוֹמוֹ. כְּֽשֶׁתִּכְנְֽנוּ, בִּשְׁנוֹת הַשִּׁבְעִים, לַהֲרֹס כִּמְעַט אֶת הַכֹּל, הוּא הָיָה כִּמְעַט הַבִּנְיָן הַיָּחִיד שֶׁנּוֹעַד לְֽהִשָּׁאֵר.

וְֽעַכְשָׁו, הַסִּפּוּר הָאָהוּב עָלַי. סְתָו אֶ֫לֶף שְׁמוֹנֶה מֵאוֹת תִּשְׁעִים וּשְׁמוֹנֶה. בִּנְיָמִין זְֽאֵב הֶרְצְל מַגִּ֫יעַ לִירוּשָׁלַ֫יִם, בְּֽדִיּוּק בְּֽאוֹתוֹ זְמַן שֶׁבּוֹ מַגִּ֫יעַ הַקֵּיסָר הַגֶּרְמָנִי, וִילְהֶלְם הַשֵּׁנִי. פָּמַלְיַת הַקֵּיסָר תּוֹפֶסֶת אֶת הַחֶ֫דֶר שֶׁלּוֹ בַּמָּלוֹן. חוֹזֵה הַמְּֽדִינָה, בְּלִי חֶ֫דֶר.

וְזֶה עוֹד לֹא הַכֹּל. בַּעַל הַמָּלוֹן, לִיפְּמָן, מְֽרַחֵם עָלָיו וּמַשְׁכִּיב אוֹתוֹ… בַּמִּסְדְּרוֹן. אָז שׁוֹמֵ֫עַ עַל כָּךְ מִיכָאֵל שְׁטֶרְן, בְּנוֹ שֶׁל יְֽהוּדָה שְׁטֶרְן, וּמַזְמִין אֶת הֶרְצְל לְֽבֵית הַמִּשְׁפָּחָה בִּרְחוֹב מָמִילָא. הֶרְצְל נִשְׁאָר שָׁם אַרְבָּעָה לֵילוֹת, וְֽהַבַּ֫יִת הוֹפֵךְ לַבָּסִיס שֶׁלּוֹ בִּירוּשָׁלַ֫יִם.

הַבַּ֫יִת הַזֶּה נִבְנָה בִּשְׁנוֹת הַשִּׁבְעִים שֶׁל הַמֵּאָה הַתְּשַׁע־עֶשְׂרֵה, בִּשְׁבִיל מִשְׁפַּחַת שְׁטֶרְן, שֶׁהִגִּיעָה מִגֶּרְמַנְיָה. וְֽהוּא כִּמְעַט נֶעֱלַם. הוּצָא עָלָיו צַו הֲרִיסָה. בִּשְׁנַת אֶ֫לֶף תְּשַׁע מֵאוֹת שִׁבְעִים וְֽאַחַת הוּא נִצַּל מֵהַדַּחְפּוֹרִים אַחֲרֵי מְֽחָאָה שֶׁל מִשְׁפַּחַת שְׁטֶרְן. וּבִשְׁנַת אֶ֫לֶף תְּשַׁע מֵאוֹת וְתִשְׁעִים הִתְעָרֵב בֵּית הַמִּשְׁפָּט הָעֶלְיוֹן וְֽעִכֵּב אֶת הַהֲרִיסָה.

בַּסּוֹף פֵּרְקוּ אוֹתוֹ אֶ֫בֶן אַחַר אֶ֫בֶן. כׇּל אֶ֫בֶן קִבְּלָה מִסְפָּר, הַכֹּל אֻחְסַן כְּֽעֶ֫שֶׂר שָׁנִים, וְֽהַבַּ֫יִת הֻרְכַּב מֵחָדָשׁ, קָרוֹב מְֽאוֹד, עַל הַשְּׂדֵרָה. אִם תִּסְתַּכְּלוּ סְבִיב הַחַלּוֹנוֹת וְֽהַדְּלָתוֹת, תִּרְאוּ מִסְפָּרִים צְבוּעִים עַל הָאֲבָנִים. אֵ֫לֶּה סִימְנֵי הַהַרְכָּבָה.

קְצָת הָלְאָה עוֹמֵד בַּ֫יִת שֶׁל אֶ֫בֶן אֲדֻמָּה וּלְֽבָנָה: בֵּית קְלַארְק. הוּא נִבְנָה בְּֽאֶ֫לֶף שְׁמוֹנֶה מֵאוֹת תִּשְׁעִים וּשְׁמוֹנֶה, בְּֽאוֹתָהּ שָׁנָה שֶׁל בִּקּוּר הֶרְצְל. בַּעַל הַבַּ֫יִת, הֶ֫רְבֶּרְט קְלַארְק, הִגִּ֫יעַ לָאָרֶץ בְּֽגִיל תֵּ֫שַׁע, עִם קְבוּצָה שֶׁל פְּרוֹטֶסְטַנְטִים אֲמֶרִיקָאִים אֲדוּקִים שֶׁהֵקִימָה אֶת הַמּוֹשָׁבָה הָאֲמֶרִיקָאִית בְּֽיָפוֹ. מְֽאֻחָר יוֹתֵר נִהֵל אֶת הַסּוֹכְנוּת הַמְּֽקוֹמִית שֶׁל תּוֹמָס קוּק, חֶבְרַת הַנְּֽסִיעוֹת הַגְּדוֹלָה שֶׁל הַתְּקוּפָה.

וְֽהוּא זֶה שֶׁאִרְגֵּן חֵ֫לֶק גָּדוֹל מֵהַנְּסִיעוֹת שֶׁל הֶרְצְל… וְשֶׁל הַקֵּיסָר, בְּֽאוֹתוֹ שָׁבוּ֫עַ. הַקֵּיסָר, שֶׁפָּמַלְיָתוֹ לָקְחָה לְֽהֶרְצְל אֶת הַחֶ֫דֶר, וְֽהֶרְצְל עַצְמוֹ, נִעְזְֽרוּ בְּֽאוֹתוֹ סוֹכֵן נְֽסִיעוֹת. הַבַּ֫יִת, עִם הָאֶ֫בֶן הָאֲדֻמָּה וְֽהַלְּבָנָה בְּֽסִגְנוֹן מַמְלוּכִי וְֽהַמִּרְפָּסוֹת מִשַּׁיִשׁ קָרָרָה, עוֹמֵד עַד הַיּוֹם, מְֽשֻׁפָּץ בִּמְקוֹמוֹ.

וְֽאָז מַגִּיעָה הַמֵּאָה הָעֶשְׂרִים, וְֽהַסִּפּוּר מִסְתַּבֵּךְ. בַּשְּׁנַ֫יִם בְּֽדֶצֶ֫מְבֶּר אֶ֫לֶף תְּשַׁע מֵאוֹת אַרְבָּעִים וָשֶׁ֫בַע, מִיָּד אַחֲרֵי הַהַצְבָּעָה בָּאֻמּוֹת הַמְּֽאֻחָדוֹת עַל תׇּכְנִית הַחֲלֻקָּה, מִתְפָּרְעִים מַצִּיתִים חֲנֻיּוֹת שֶׁל יְֽהוּדִים וּבֵית קוֹלְנוֹ֫עַ בְּֽבַעֲלוּת עֲרָבִית. בְּֽאֶ֫לֶף תְּשַׁע מֵאוֹת אַרְבָּעִים וּשְׁמוֹנֶה הוֹפֶכֶת הַשְּׁכוּנָה לִשְׂדֵה קְרָב, וְתוֹשָׁבֶ֫יהָ, יְֽהוּדִים וַעֲרָבִים, בּוֹרְחִים.

בְּֽעֶשְׂרִים וּשְׁנַ֫יִם בְּֽמַאי אוֹתָהּ שָׁנָה, הַקּוֹנְסוּל הַכְּלָלִי שֶׁל אַרְצוֹת הַבְּרִית, תּוֹמָס וָסוֹן, חוֹזֵר מִיְּֽשִׁיבָה שֶׁל וַעֲדַת הַהֲפוּגָה שֶׁל הָאֻמּוֹת הַמְּֽאֻחָדוֹת. כַּדּוּר שֶׁל צַלָּף פּוֹגֵ֫עַ בּוֹ, וְֽהוּא מֵת לְֽמׇחֳרָת. מִי יָרָה? אַף אֶחָד לֹא יוֹדֵ֫עַ. הַיּוֹרֶה לֹא זֻהָה מֵעוֹלָם, וְֽהָעֵדֻיּוֹת סוֹתְרוֹת זוֹ אֶת זוֹ.

אַחֲרֵי הֶסְכֵּם שְׁבִיתַת הַנֶּ֫שֶׁק, הַגְּבוּל עִם יַרְדֵּן עוֹבֵר בְּֽאֶמְצַע הַשְּׁכוּנָה. שְׁלוֹשָׁה רְֽבָעִים מִמֶּנָּה בַּצַּד הַיִּשְׂרְֽאֵלִי, וְֽהָרֶ֫בַע הַמִּזְרָחִי הוֹפֵךְ לִשְׁטַח הֶפְקֵר. חוֹמַת בֶּטוֹן עָבָה חוֹצָה אֶת הַשְּׁכוּנָה, וּמִן הַחוֹמוֹת יוֹרִים צַלָּפִים יַרְדֵּנִים, וְֽאַף זוֹרְקִים אֲבָנִים. גָּרִים כָּאן בְּֽעִקָּר עוֹלִים מִכּוּרְדִיסְטָן וִילָדֵיהֶם, וּלְֽאַט לְֽאַט נִכְנָסִים מוּסָכִים וְתַעֲשִׂיָּה קַלָּה.

בְּֽאֶ֫לֶף תְּשַׁע מֵאוֹת שִׁשִּׁים וָשֶׁ֫בַע הַגְּדֵרוֹת נוֹפְלוֹת. בְּֽאֶ֫לֶף תְּשַׁע מֵאוֹת שִׁבְעִים וּשְׁתַּ֫יִם קוֹבַעַת תׇּכְנִית, בְּֽהִשְׁתַּתְּפוּת הָאַדְרִיכָל מֹשֶׁה סַפְדִיָּה, לַהֲרֹס כִּמְעַט אֶת הַכֹּל, חוּץ מֵהַמִּנְזָר.

וְזֶה הַחֵ֫לֶק שֶׁשּׁוֹכְחִים. חֶבְרַת קַרְתָּא מְֽפַנָּה כִּשְׁבַע מֵאוֹת מִשְׁפָּחוֹת, רֻבָּן עוֹלִים יְֽהוּדִים מֵאַרְצוֹת עֲרָב. הַכַּ֫עַס שֶׁלָּהֶם מְֽזִין אֶת תְּנוּעַת הַפַּנְתֵּרִים הַשְּֽׁחוֹרִים בִּשְׁנוֹת הַשִּׁבְעִים. הַפִּנּוּי מִסְתַּיֵּם בְּֽאֶ֫לֶף תְּשַׁע מֵאוֹת שְׁמוֹנִים וּשְׁמוֹנֶה. מֵאוֹתוֹ רֶ֫גַע, מָמִילָא כְּבָר אֵינָהּ שְׁכוּנָה.

בִּזְמַן הַהֲרִיסָה נִזְהֲרוּ הַפּוֹעֲלִים לֹא לִשְׁבֹּר אֶת הָאֲבָנִים, כִּי רַבּוֹת מֵהֶן נוֹעֲדוּ לְֽהַרְכָּבָה מֵחָדָשׁ שֶׁל בִּנְיָנִים בַּשְּׂדֵרָה הַחֲדָשָׁה. עַד הַיּוֹם, כַּחֲצִי תְּרֵיסַר בִּנְיָנִים הִיסְטוֹרִיִּים מַרְאִים אֶת הָאֲבָנִים הַמְּֽמֻסְפָּרוֹת. הַחֵ֫לֶק הָרִאשׁוֹן שֶׁל הַשְּׂדֵרָה נִפְתַּח בְּֽעֶשְׂרִים וּשְׁמוֹנָה בְּֽמַאי אַלְפַּ֫יִם וְשֶׁ֫בַע: כְּֽשֵׁשׁ מֵאוֹת מֶ֫טֶר שֶׁל טַיֶּ֫לֶת, בֵּין הָעִיר הַחֲדָשָׁה לְֽשַׁ֫עַר יָפוֹ.

וְנִגְמֹר בְּֽשַׁ֫עַר יָפוֹ. הִסְתַּכְּלוּ עַל הַפִּרְצָה בַּחוֹמָה, לְֽיַד הַשַּׁ֫עַר הַיָּשָׁן, שָׁם עוֹבְרוֹת הַמְּֽכוֹנִיּוֹת. הִיא נִפְתְּֽחָה רַק בְּֽאֶ֫לֶף שְׁמוֹנֶה מֵאוֹת תִּשְׁעִים וּשְׁמוֹנֶה, כְּֽדֵי שֶׁכִּרְכָּרוֹת יוּכְלוּ לְֽהִכָּנֵס לָעִיר. וּבְֽאֶ֫לֶף תְּשַׁע מֵאוֹת וּשְׁמוֹנֶה בָּנוּ הָעוֹתְמָאנִים מִגְדַּל שָׁעוֹן מֵעַל הַשַּׁ֫עַר. הַבְּרִיטִים הֵסִירוּ אוֹתוֹ כְּֽעֶ֫שֶׂר שָׁנִים אַחַר כָּךְ.

אֶ֫לֶף שְׁמוֹנֶה מֵאוֹת תִּשְׁעִים וּשְׁמוֹנֶה, שׁוּב: הַשָּׁנָה שֶׁל הֶרְצְל, שֶׁל הַקֵּיסָר וְשֶׁל בֵּית קְלַארְק. שָׁנָה גְּדוֹלָה לְֽמָמִילָא.

אָז בַּפַּעַם הַבָּאָה שֶׁאַתֶּם מְֽטַיְּלִים בַּשְּׂדֵרָה, חִשְׁבוּ עַל הַבְּרֵכָה מִימֵי הוֹרְדוֹס, עַל הַקַּפֶּ֫לָּה שֶׁמִּתַּחַת לַחַנְיוֹן, עַל הַנְּֽזִירוֹת וְֽהַחֲנֻיּוֹת שֶׁלָּהֶן, עַל הֶרְצְל בַּמִּסְדְּרוֹן… וְֽעַל הַמִּשְׁפָּחוֹת שֶׁגָּרוּ כָּאן לֹא מִזְּמַן. וְֽהִסְתַּכְּלוּ עַל הַמִּסְפָּרִים שֶׁעַל הָאֲבָנִים.

הָיָה זֶה הָאֲבָנִים מְֽדַבְּרוֹת. לְֽהִתְרָאוֹת.
