D|בְּרוּכִים הַבָּאִים לְ„הָאֲבָנִים מְדַבְּרוֹת”. הַיּוֹם אֲנַחְנוּ מְטַיְּילִים בִּירוּשָׁלַיִם, בַּשְׂדֵרָה שֶׁכֻּולָּם מַכִּירִים: מִמֵּילָא. הַחֲנוּיוֹת, בָּתֵּי הַקָּפֶה, הַנּוֹף אֶל הַחוֹמוֹת……… וּבְכׇל זֹאת, כִּמְעַט אַף אֶחָד לֹא יוֹדֵעַ מָה הָיָה כָּאן קֹודֶם.
C|אֲנִי עָבַרְתִּי כָּאן מֵאָה פְּעָמִים, בִּשְׁבִיל קָפֶה אוֹ בַּדֶּרֶךְ לְשַׁעַר יָפוֹ. אָז תַּגִּיד לִי: מָה הָיְיתָה מִמֵּילָא לִפְנֵי הַקַּנְיוֹן?
D|שְׁכוּנָה אֲמִיתִּית. עִם בָּתִּים, סַדְנָאוֹת, סוֹחֲרִים וּמִשְׁפָּחוֹת. וְאִם מְגָרְדִים קְצָת, מַגִּיעִים הַרְבֵּה יוֹתֵר רָחוֹק: עַד הַתְּקוּפָה הָרוֹמִית, וַאֲפִילּוּ לִפְנֵי כֵן.
C|אָז נַתְחִיל מֵהֲכִי עַתִּיק.
D|נַתְחִיל מֵהַמַּיִם. מַמָּשׁ מִמַּעֲרָב לַשְּׂדֵרָה, בְּגַן הָעַצְמָאוּת, יֵשׁ בְּרֵיכָה גְּדוֹלָה: בְּרֵיכַת מִמֵּילָא. זֶה מַאֲגַר מַיִם עַתִּיק, כַּנִּרְאֶה מִיְּמֵי הוֹרְדוּס. הוּא הֶחֱזִיק בְּעֵרֶךְ שְׁלוֹשִׁים אֶלֶף מֶטֶר מְעֻוקָּב שֶׁל מַיִם, וְתְעָלָה תִּתְקַרְקַעִית חִיבְּרָה אוֹתוֹ לִבְרֵיכַת חִזְקִיָּהוּ, בְּתוֹךְ הָעִיר הָעַתִּיקָה.
C|כְּלוֹמַר לִפְנֵי חַלּוֹנוֹת הָרַאֲוָוה, הָיוּ כָּאן מַיִם לְכׇל הָעִיר.
D|בְּדִיּוּק. וּסְבִיב הַבְּרֵיכָה הַזֹּאת מִתְרַחֵשׁ אַחַד הַסִּיפּוּרִים הַקָּשִׁים שֶׁל הַמָּקוֹם. בִּשְׁנַת שֵׁשׁ מֵאוֹת אַרְבעעשְׂרֵה הַפָּרְסִים כּוֹבְשִׁים אֶת יְרוּשָׁלַיִם. נוֹצְרִים נִטְבָּחִים לְיַד הַבְּרֵיכָה. כְּרוֹנִיקָן בִּיזַנְטִי בְּשֵׁם אַסְטְרָטֶגְיּוּס מְתָאֵר אֶת הַמִּקְרֶה, וּמַאֲשִׁים אֶת הַיְּהוּדִים בִּנְקָמָה.
C|זֹאת הַגִּרְסָה שֶׁלּוֹ, נָכוֹן?
D|כֵּן, וְחָשׁוּב לְהַגִּיד אֶת זֶה. זֶה סִיפּוּר שֶׁל מָקוֹר אֶחָד, נָזִיר נוֹצְרִי מֵאוֹתָהּ תְּקוּפָה. אֲבָל מָה שֶׁמְּרַתֵּק הוּא שֶׁהָאַרְכֵאוֹלוֹגְיָה נָתְנָה לוֹ הֵד.
C|אֵיךְ?
D|בְּסוֹף שְׁנוֹת הַשְּׁמוֹנִים מַתְחִילִים לַחְפֹּור בִּשְׁבִיל הַחַנְיוֹן שֶׁל הַקַּנְיוֹן הָעֲתִידִי. וְהַפּוֹעֲלִים נִתְקָלִים בְּמְעָרָה טִבְעִית, בְּאֹורֶךְ שֶׁל כִּשְׁתֵּיםעֶשְׂרֵה מֶטֶר, מְלֵאָה בַּעֲצָמוֹת. מְאוֹת אֲנָשִׁים. רַבִּים מֵהֶם צְעִירִים, גְּבָרִים וְנָשִׁים.
C|מִתַּחַת לַחַנְיוֹן?
D|מִתַּחַת לַחַנְיוֹן. לִפְנֵי הַמְּעָרָה עָמְדָה קִפְלָה קְטַנָּה, עִם רִצְפַּת פְּסֵיפָס, וּבאַפְסִיס שְׂרִידֵי צִיּוּר: מַלְאָךְ שֶׁפּוֹרֵשׂ אֶת יָדָיו. הָאַרְכֵאוֹלוֹג שֶׁנִּיהֵל אֶת הַחֲפִירָה, רוֹנִי רַיְיךְ, הִמְשִׁיךְ לַעֲבֹוד לַמְרוֹת הַטְרָדוֹת שֶׁל מַפְגִּינִים חֲרֵדִים. לְדַעְתּוֹ, אֵלֶּה כַּנִּרְאֶה הַקׇּורְבָּנוֹת שֶׁל שֵׁשׁ מֵאוֹת אַרְבַּעעשְׂרֵה.
C|וְאֶפְשָׁר לִרְאוֹת הַיּוֹם אֶת הַקַּפְלָה?
D|לֹא. הִיא מִתַּחַת לַחַנְיוֹן. כְּשֶׁאַתְּ מְחַנָּה אֶת הָרֶכֶב בְּמְמֵּילָא, אוּלַי אַתְּ עוֹמֶדֶת בְּדִיּוּק מֵעָלֶיהָ.
C|זֶה מְצַמְרֵר. נַעֲלֶה בַּחֲזָרָה לְמַעְלָה?
D|נִקְפֹּוץ לְסוֹף הַמֵּאָה הַתְּשְׁעעשְׂרֵה. אָז הַמָּקוֹם כִּמְעַט רֵיק: בְּעִיקָּר צֹומֶת דְּרָכִים, מוּל שַׁעַר יָפוֹ. וְאַחַד הַבִּנְיָינִים הָרִאשׁוֹנִים הוּא מִנְזָר: שֶׁל הָאֲחָיוֹת בְּנוֹת הַצְּדָקָה שֶׁל סַן וַנְסַן דֶּה פּוֹל, שֶׁהִגִּיעוּ בַּהַתְחָלָה כְּדֵי לְטַפֵּל בְּחוֹלֵי צָרַעַת.
C|נְזִירוֹת בְּאֶמְצַע שְׁכוּנָה מִסְחָרִית?
D|הֵן בִּכְלָל אֵלֶּה שֶׁהִתְחִילוּ אֶת הַמִּסְחָר! הָאֲחָיוֹת בָּנוּ שׁוּרַת חֲנוּיוֹת מוּל הַמִּנְזָר, וּבְכֶסֶף מִדְּמֵי הַשְּׂכִירוּת הֵן הֶחֱזִיקוּ אֶת הַמִּנְזָר.
C|אָז חֲנוּיוֹת הַיֻּוקְרָה שֶׁל הַיּוֹם הֵן קְצָת הַיּוֹרְשׁוֹת שֶׁל הָרַעְיוֹן שֶׁלָּהֶן.
D|בְּמִידָּה מְסֻויֶּמֶת, כֵּן. וְהַמִּנְזָר עֲדַיִין עוֹמֵד בִּמְקוֹמוֹ. כְּשֶׁרָצוּ לַהֲרֹוס הַכֹּול בִּשְׁנוֹת הַשִּׁבְעִים, הוּא הָיָה כִּמְעַט הַבִּנְיָין הַיָּחִיד שֶׁתֻּוכְנַן לְהִישָּׁאֵר.
C|נַחֲזֹור לָזֶה. שָׁמַעְתִּי שֶׁהֵרְצְל עָבַר כָּאן?
D|זֶה הַסִּיפּוּר הָאָהוּב עָלַיי. סְתָיו אֶלֶף שְׁמוֹנֶה מֵאוֹת תִּשְׁעִים וּשְׁמוֹנֶה. תֵּאוֹדוֹר הֵרְצְל מַגִּיעַ לִירוּשָׁלַיִם……… בְּדִיּוּק בְּאוֹתוֹ זְמַן כְּמוֹ הַקֵּיסָר הַגֶּרְמָנִי, וִילְהֵלְם הַשֵּׁנִי. וְהַפָּמַלְיָה שֶׁל הַקֵּיסָר תּוֹפֶסֶת אֶת הַחֶדֶר שֶׁלּוֹ בַּמָּלוֹן.
C|חוֹזֵה הַמְּדִינָה, בְּלִי חֶדֶר?
D|יוֹתֵר גָּרוּעַ: בַּעַל הַמָּלוֹן, אֶחָד בְּשֵׁם לִיפְמָן, מְרַחֵם עָלָיו וּמַשְׁכִּיב אוֹתוֹ……… בַּמִּסְדְּרוֹן. וְאָז תּוֹשַׁב הַשְּׁכוּנָה, מִיכָאֵל שְׁטֵרְן, שׁוֹמֵעַ עַל זֶה. הוּא מַזְמִין אֶת הֵרְצְל לְבֵית הַמִּשְׁפָּחָה, בָּרְחוֹב מִמֵּילָא. הֵרְצְל נִשְׁאַר שָׁם אַרְבָּעָה לֵילוֹת, וְהַבַּיִת הוֹפֵךְ לַבָּסִיס שֶׁלּוֹ בִּזְמַן הַבִּיקּוּר.
C|וְהַבַּיִת הַזֶּה עוֹד קַיָּים?
D|כֵּן, אֲבָל הוּא כִּמְעַט נֶעֱלַם. הוֹצִיאוּ עָלָיו צַו הֲרִיסָה. בִּשְׁנַת אֶלֶף תֵּשַׁע מֵאוֹת שִׁבְעִים וְאַחַת, יוֹמַן חֲדָשׁוֹת שֶׁל אוֹתָהּ תְּקוּפָה מַרְאֶה שֶׁהוּא נִיצַּל מֵהַדַּחְפּוּרִים אַחֲרֵי מְחָאָה שֶׁל מִשְׁפַּחַת שְׁטֵרְן. וְאָז, בִּשְׁנַת אֶלֶף תֵּשַׁע מֵאוֹת תִּשְׁעִים, בֵּית הַמִּשְׁפָּט הָעֶלְיוֹן מִתְעָרֵב וּמְעַכֵּב אֶת הַהֲרִיסָה.
C|וּבַסּוֹף?
D|בַּסּוֹף פֵּירְקוּ אוֹתוֹ אֶבֶן אַחַר אֶבֶן. כׇּל אֶבֶן קִיבְּלָה מִסְפָּר, וְהַכֹּול אֻוחְסַן כְּעֶשֶׂר שָׁנִים, וְאָז הֻורְכַּב מֵחָדָשׁ קָרוֹב מְאוֹד, עַל הַשִּׂדְרָה.
C|רֶגַע……… הַמִּסְפָּרִים שֶׁרוֹאִים צְבוּעִים עַל חֵלֶק מֵהָאֲבָנִים, זֶה זֶה?
D|זֶה זֶה. אֵלֶּה סִימָנֵי הַרְכָּבָה. בַּפַּעַם הַבָּאָה שֶׁאַתְּ עוֹבֶרֶת שָׁם, תִּסְתַּכְּלִי מִסָּבִיב לַחַלּוֹנוֹת וְלַדְּלָתוֹת.
C|יֵשׁ עוֹד בַּיִת, קְצָת יוֹתֵר רָחוֹק, עִם אֲבָנִים אֲדֻומּוֹת וּלְבָנוֹת………
D|בֵּית קְלַארְק. נִבְנָה בִּשְׁנַת אֶלֶף שְׁמוֹנֶה מֵאוֹת תִּשְׁעִים וּשְׁמוֹנֶה, בְּאוֹתָהּ שָׁנָה שֶׁל בִּיקּוּר הֵרְצְל. הַבְּעָלִים, הֶרְבֵּרְט קְלַארְק, הִגִּיעַ לְאֶרֶץ הַקֹּודֶשׁ בְּגִיל תֵּשַׁע, עִם קְבוּצָה שֶׁל פְּרוֹטֶסְטַנְטִים אָמֵרִיקָאִים אֲדוּקִים. מְאֻוחָר יוֹתֵר הוּא מְנַהֵל אֶת הַסּוֹכְנוּת הַמְּקוֹמִית שֶׁל תּוֹמָס קוּק, חֶבְרַת הַנְּסִיעוֹת הַגְּדוֹלָה שֶׁל הַתְּקוּפָה.
C|וְאָז?
D|וְאָז הוּא זֶה שֶׁמְּאַרְגֵּן חֵלֶק גָּדוֹל מֵהַנְּסִיעוֹת שֶׁל הֵרְצְל……… וְשֶׁל הַקֵּיסָר, בְּאוֹתוֹ שָׁבוּעַ!
C|שְׁנֵי הָאֲנָשִׁים שֶׁהִתְוַוכְּחוּ עַל חֶדֶר בַּמָּלוֹן, הָיָה לָהֶם אוֹתוֹ סוֹכֵן נְסִיעוֹת?
D|בְּדִיּוּק. וְהַבַּיִת שֶׁלּוֹ, עִם הָאֶבֶן הָאֲדֻומָּה וְהַלְּבָנָה בְּסִגְנוֹן מַמְלוּכִּי וְהַמִּרְפָּסוֹת מִשִּׁיִשׁ קְרָרָה, עֲדַיִין עוֹמֵד שָׁם, מְשֻׁופָּץ בִּמְקוֹמוֹ.
C|מַגִּיעִים לַמֵּאָה הָעֶשְׂרִים. אֲנִי מַנִּיחָה שֶׁזֶּה מִסְתַּבֵּךְ.
D|מְאוֹד. בַּשֵּׁנִי בְּדֵצֶמְבֶּר אֶלֶף תֵּשַׁע מֵאוֹת אַרְבָּעִים וְשֶׁבַע, מִיָּד אַחֲרֵי הַהַצְבָּעָה בָּאוּ״ם עַל תׇּוכְנִית הַחֲלֻוקָּה, מִתְפָּרְעִים מַצִּיתִים חֲנוּיוֹת שֶׁל יְהוּדִים וְקוֹלְנוֹעַ בְּבַעֲלוּת עַרְבִית. בִּשְׁנַת אֶלֶף תֵּשַׁע מֵאוֹת אַרְבָּעִים וּשְׁמוֹנֶה הַשְּׁכוּנָה הוֹפֶכֶת לִשְׂדֵה קְרָב, וְהַתּוֹשָׁבִים, יְהוּדִים וְעַרְבִים, בּוֹרְחִים.
C|יֵשׁ גַּם אֶת הַסִּיפּוּר שֶׁל הַקּוֹנְסוּל הָאָמֵרִיקָאִי, לֹא?
D|כֵּן. בְּעֶשְׂרִים וּשְׁנַיִים בְּמַאי אֶלֶף תֵּשַׁע מֵאוֹת אַרְבָּעִים וּשְׁמוֹנֶה, הַקּוֹנְסוּל הַכְּלָלִי שֶׁל אַרְצוֹת הַבְּרִית, תוֹמָס וָסוֹן, חוֹזֵר מִיְּשִׁיבָה שֶׁל וַעֲדַת הַהֲפוּגָה שֶׁל הָאוּ״ם. כַּדּוּר שֶׁל צַלָּף פּוֹגֵעַ בּוֹ, וְהוּא מֵת לְמׇחֳרָת.
C|יוֹדְעִים מִי יָרָה?
D|לֹא, וְזֶה כׇּל הָעִנְיָין: הַיּוֹרֶה לֹא זֹוהָה אַף פַּעַם, וְהָעֵדוּיוֹת סוֹתְרוֹת. אָז מְסַפְּרִים אֶת הָעֻובְדָּה, בְּלִי לְהַצְבִּיעַ עַל אָשֶׁם.
C|וְאַחֲרֵי הַמִּלְחָמָה?
D|אַחֲרֵי שְׁבִיתַת הַנֶּשֶׁק, הַגְּבוּל עִם יַרְדֵּן עוֹבֵר בְּאֶמְצַע הַשְּׁכוּנָה. שְׁלוֹשָׁה רְבָעִים בַּצַּד הַיִּשְׂרְאֵלִי, וְהָרֶבַע הַמִּזְרָחִי הוֹפֵךְ לְשֶׁטַח הֶפְקֵר. חוֹמַת בֵּטוֹן עָבָה חוֹצָה אֶת הַשְּׁכוּנָה, וּמֵהַחוֹמוֹת צַלָּפִים יַרְדֵּנִים יוֹרִים לְעֵבֶר הָאֵזוֹר. אֲפִילּוּ זוֹרְקִים אֲבָנִים.
C|וּמִי גָּר שָׁם בַּזְּמַן הַזֶּה?
D|בְּעִיקָּר עוֹלִים מִכּוּרְדִּיסְטָן וְהַיְּלָדִים שֶׁלָּהֶם. וּלְאַט לְאַט נִכְנָסִים מוּסָכִּים וְתַעֲשִׂיָּיה קַלָּה.
C|וְאֶלֶף תֵּשַׁע מֵאוֹת שִׁישִּׁים וְשֶׁבַע מְשַׁנֶּה הַכֹּול.
D|בְּשִׁישִּׁים וְשֶׁבַע הַגְּדָּרוֹת נוֹפְלוֹת. וְאוֹמְרִים: אֶת הַשְּׁכוּנָה הַזֹּאת, הַצְּמוּדָה לָעִיר הָעַתִּיקָה, צָרִיךְ לְשַׁנּוֹת. בִּשְׁנַת אֶלֶף תֵּשַׁע מֵאוֹת שִׁבְעִים וּשְׁתַּיִים, תׇּוכְנִית עִם הָאַדְרִיכָל מֹשֶׁה סֶפְדִי קוֹבַעַת לַהֲרֹוס כִּמְעַט הַכֹּול, חוּץ מֵהַמִּנְזָר.
C|וְהַתּוֹשָׁבִים?
D|זֶה הַחֵלֶק שֶׁשּׁוֹכְחִים הַרְבֵּה פְּעָמִים. חֶבְרַת קַרְתָּא מְפְנָה בְּעֵרֶךְ שְׁבַע מֵאוֹת מִשְׁפָּחוֹת, רֻובָּן שֶׁל עוֹלִים יְהוּדִים מֵאַרְצוֹת עֲרַב. הַכַּעַס שֶׁלָּהֶם מֵזִין אֶת תְּנוּעַת הַפַּנְתָּרִים הַשְּׁחֹורִים בִּשְׁנוֹת הַשִּׁבְעִים. הַפִּינּוּי מִסְתַּיֵּים בִּשְׁנַת אֶלֶף תֵּשַׁע מֵאוֹת שְׁמוֹנִים וּשְׁמוֹנֶה. בַּיּוֹם הַזֶּה, מִמֵּילָא מַפְסִיקָה לִהְיוֹת שְׁכוּנָה.
C|מְטֹורָף לַחְשֹׁוב עַל כׇּל זֶה כְּשֶׁיּוֹשְׁבִים לְקָפֶה בַּשְׂדְרָה.
D|וּבִזְמַן הַהֲרִיסָה הַפּוֹעֲלִים הָיוּ צְרִיכִים לְהִיזָּהֵר לֹא לִשְׁבֹּור אֶת הָאֲבָנִים, כִּי רַבּוֹת מֵהֶן נוֹעֲדוּ לְהַרְכָּבָה מֵחָדָשׁ שֶׁל הַבִּנְיָינִים בַּשְּׂדֵרָה הַחֲדָשָׁה. הַיּוֹם, כַּחֲצִי תְּרֵיסַר בִּנְיָינִים הִיסְטוֹרִיִּים עֲדַיִין מַרְאִים אֶת הָאֲבָנִים הַמְּמֻוסְפָּרוֹת.
C|וְהַשְּׂדֵרָה עַצְמָהּ נִפְתַּחַת מָתַי?
D|הַחֵלֶק הָרִאשׁוֹן נִפְתַּח בְּעֶשְׂרִים וּשְׁמוֹנֶה בְּמַאי אַלְפַּיִים וְשֶׁבַע. שֵׁשׁ מֵאוֹת מֶטֶר שֶׁל טַיֶּילֶת, בֵּין הָעִיר הַחֲדָשָׁה לְשַׁעַר יָפוֹ.
C|אָז מְסַיְּימִים בִּשַׁעַר יָפוֹ?
D|מְסַיְּימִים שָׁם. תִּסְתַּכְּלִי עַל הַפְּרְצָה בַּחוֹמָה, לְיַד הַשַּׁעַר הַיָּשָׁן, אֵיפֹה שֶׁעוֹבְרוֹת הַמְּכוֹנִיּוֹת. הִיא נִפְתְּחָה רַק בִּשְׁנַת אֶלֶף שְׁמוֹנֶה מֵאוֹת תִּשְׁעִים וּשְׁמוֹנֶה, כְּדֵי שֶׁכַּרְכָּרוֹת יוּכְלוּ לְהִיכָּנֵס לָעִיר. וּבִשְׁנַת אֶלֶף תֵּשַׁע מֵאוֹת וּשְׁמוֹנֶה הָעוֹתְ׳מָאנִים בּוֹנִים מִגְדַּל שָׁעוֹן מֵעַל הַשַּׁעַר. הַבְּרִיטִים מוֹרִידִים אוֹתוֹ כְּעֶשֶׂר שָׁנִים אַחַר כָּךְ.
C|אֶלֶף שְׁמוֹנֶה מֵאוֹת תִּשְׁעִים וּשְׁמוֹנֶה, שׁוּבַ! הַשָּׁנָה שֶׁל הֵרְצְל, שֶׁל הַקֵּיסָר וְשֶׁל בֵּית קְלַארְק.
D|שָׁנָה גְּדוֹלָה מְאוֹד לְמִמֵּילָא. אָז בַּפַּעַם הַבָּאָה שֶׁאַתֶּם הוֹלְכִים בַּשְּׂדֵרָה הַזֹּאת, תַּחְשְׁבוּ עַל הַבְּרֵיכָה שֶׁל הוֹרְדּוּס, עַל הַקַּפָּלָה מִתַּחַת לַחַנְיוֹן, עַל הָאֲחָיוֹת וְהַחֲנוּיוֹת שֶׁלָּהֶן, עַל הֵרְצְל בַּמִּסְדְּרוֹן……… וְעַל הַמִּשְׁפָּחוֹת שֶׁגָּרוּ כָּאן לֹא מִזְּמַן.
C|וְתִסְתַּכְּלוּ עַל הַמִּסְפָּרִים שֶׁעַל הָאֲבָנִים.
D|בְּעִיקָּר עַל הַמִּסְפָּרִים שֶׁעַל הָאֲבָנִים. זֶה הָיָה „הָאֲבָנִים מְדַבְּרוֹת”. לְהִתְרָאוֹת.
